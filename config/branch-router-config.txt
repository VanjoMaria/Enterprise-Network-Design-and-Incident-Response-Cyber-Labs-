! =====================================================================
! Cyber Labs — Branch Router (1941)
! Branch LAN gateway + DHCP server; dual-homed to HQ (primary + backup WAN)
! =====================================================================

hostname BranchRouter
!
! NOTE: "no ip cef" / "no ipv6 cef" retained deliberately — matches the
! platform-specific setting used on Edge Router in this Packet Tracer build.
no ip cef
no ipv6 cef
!
service timestamps log datetime msec
!
ip domain-name Cyber.labs
username admin privilege 15 secret 5 $1$mERr$5.a6P4JqbNiMX01usIfka/
!
ip ssh version 2
line con 0
line aux 0
line vty 0 4
 login local
 transport input ssh
line vty 5 15
 login local
 transport input ssh
!
spanning-tree mode pvst
!
! ---------------------------------------------------------------------
! DHCP — Branch LAN
! ---------------------------------------------------------------------
ip dhcp relay information trust-all
!
ip dhcp excluded-address 10.20.10.1 10.20.10.5
!
ip dhcp pool BRANCHLAN
 network 10.20.10.0 255.255.255.0
 default-router 10.20.10.1
 dns-server 8.8.8.8
!
! ---------------------------------------------------------------------
! Interfaces
! ---------------------------------------------------------------------
interface GigabitEthernet0/0
 description Branch-LAN (to access switch)
 ip address 10.20.10.1 255.255.255.0
 duplex auto
 speed auto
!
interface GigabitEthernet0/1
 description Unused-spare
 no ip address
 duplex auto
 speed auto
 shutdown
!
interface Serial0/0/0
 description Link-to-EdgeRouter-Primary
 ip address 10.255.255.14 255.255.255.252
!
interface Serial0/0/1
 description Link-to-BackupWANRouter
 ip address 10.255.255.18 255.255.255.252
 clock rate 2000000
!
interface FastEthernet0/1/0
 description Unused-spare (onboard switch port, not in topology)
 switchport mode access
 shutdown
!
interface FastEthernet0/1/1
 description Unused-spare (onboard switch port, not in topology)
 switchport mode access
 shutdown
!
interface FastEthernet0/1/2
 description Unused-spare (onboard switch port, not in topology)
 switchport mode access
 shutdown
!
interface FastEthernet0/1/3
 description Unused-spare (onboard switch port, not in topology)
 switchport mode access
 shutdown
!
interface Vlan1
 no ip address
 shutdown
!
! ---------------------------------------------------------------------
! Routing — OSPF Area 0
! ---------------------------------------------------------------------
router ospf 1
 log-adjacency-changes
 passive-interface GigabitEthernet0/0
 network 10.255.255.12 0.0.0.3 area 0
 network 10.255.255.16 0.0.0.3 area 0
 network 10.20.10.0 0.0.0.255 area 0
!
ip classless
!
! ---------------------------------------------------------------------
! Monitoring
! ---------------------------------------------------------------------
snmp-server community CYBERLABS.RO RO
logging host 10.10.20.50
!
end
